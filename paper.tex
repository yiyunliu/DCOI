\section{Lemmas}

\begin{lemma}[Transitivity of indistinguishability]
  \stack{.~}\footnote{\suppl{theories}{geq.v}{ieq\_trans\_heterogeneous}}
  If $[[E |- a == b @ l0]]$ and $[[E |- b == c @ l1]]$,
  then $[[E |- a == c @ l0 /\ l1]]$
\end{lemma}

\begin{lemma}[Congruence of indistinguishability]
  \stack{.~}\footnote{\suppl{theories}{geq.v}{ieq\_morphing\_mutual}}
  Suppose $[[E |- b0 == b1 @ l l0]]$.
  If $[[E ++ x : l0 |- a0 == a1 @ l]]$,
  then $[[E |- a0 { b0 / x } == a1 { b1 / x } @ l]]$;
  and if $[[E ++ x : l0 |- a0 == a1 @ l l1]]$,
  then $[[E |- a0 { b0 / x } == a1 { b1 / x } @ l l1]]$.
\end{lemma}

\begin{lemma}[Simulation]
  \stack{.~}\footnote{\suppl{theories}{conv.v}{simulation}}
  \label{lem:simulation}
  If $[[E |- a0 == b0 @ l]]$ and $[[a0 => a1]]$,
  then there exists some $[[b1]]$ such that $[[b0 => b1]]$
  and $[[E |- a1 == b1 @ l]]$.
\end{lemma}

\begin{lemma}[Preservation of definitional equality]
  \stack{.~}\footnote{\suppl{theories}{conv.v}{iconv\_par2}}
  If $[[E |- a0 == b0]]$, $[[a0 => a1]]$, and $[[b0 => b1]]$,
  then $[[E |- a1 == b1]]$.
\end{lemma}

\begin{lemma}[Transitivity of definitional equality]
  \stack{.~}\footnote{\suppl{theories}{conv.v}{conv\_trans}}
  If $[[E |- a == b]]$ and $[[E |- b == c]]$,
  then $[[E |- a == c]]$.
\end{lemma}

\begin{lemma}[Substitutivity of definitional equality]
  \stack{.~}\footnote{\suppl{theories}{conv.v}{conv\_subst}}
  If $[[E ++ x : l |- a0 == a1]]$ and $[[E |- b : l]]$,
  then $[[E |- a0 { b / x } == a1 { b / x }]]$.
\end{lemma}

\begin{lemma}[Subsumption]
  \stack{.~}{\footnote{\jc{I couldn't find the lemma for this one?}}}
  If $[[G |- a : l0 A]]$ and $[[l0 <= l1]]$,
  then $[[G |- a : l1 A]]$.
\end{lemma}

\begin{lemma}[Well-gradedness of well-typed terms]
  \stack{.~}\footnote{\suppl{theories}{typing\_conv.v}{typing\_iok}}
  If $[[G |- a : l A]]$, then $[[| G | |- a : l]]$.
\end{lemma}

\begin{lemma}[Reflexive equality of well-typed terms]
  \stack{.~}\footnote{\suppl{theories}{typing\_conv.v}{typing\_conv}}
  If $[[G |- a : l A]]$, then $[[| G | |- a == a]]$.
\end{lemma}

\section{Notes}

The level of scrutinees must be bounded by the level of the overall eliminator.
In particular, in \rref{Wt-J}, the level of an equality proof $[[p]]$
must be bounded by that of its elimination $[[J c p]]$,
and in \rref{Wt-Let}, the level $[[l1]]$ of a pair $[[a]]$
must be bounded by that of its elimination $[[let (l0 x, y) = l1 a in b]]$.
The bound is required to prove the simulation property (\cref{lem:simulation}).
Suppose that scrutinees could instead be typed at any level.
First, the indistinguishability rules need to be updated
so that scrutinees are indistinguishable if their level is higher
than that of the overall eliminator, just like in \rref{I-App}.
The modified premises are highlighted in \hlcolor{} below.
%
\begin{mathpar}
  \setlength{\fboxsep}{1.5pt}
  \infer[I-J-bad]{
    \mathcolorbox{\hlcolor}{[[E |- p0 == p1 @ l l0]]} \\
    [[E |- c0 == c1 @ l]]
  }{
    [[E |- J c0 l0 p0 == J c1 l0 p1 @ l]]
  }
  \and
  \infer[I-Let-bad]{
    \mathcolorbox{\hlcolor}{[[E |- a0 == a1 @ l l1]]} \\
    [[(# E ++ x : l0 #) ++ y : l1 |- b0 == b1 @ l]]
  }{
    [[E |- let (l0 x, y) = l1 a0 in b0 == let (l0 x, y) = l1 a1 in b1 @ l]]
  }
\end{mathpar}

As concrete examples, by these alternate rules,
supposing $[[J c p]]$ were well typed in context $[[G]]$
at level $[[L]]$ with $[[G |- p : H a ~ L a]]$,
we can conclude that $[[| G | |- J c refl == J c p @ L]]$.
The problem is that the left-hand side reduces to $[[c]]$,
while the right-hand side does not necessarily reduce to something indistinguishable from $[[c]]$,
thereby violating the simulation property.
The same issue arises with pairs, where we may be able to conclude that
$[[| G | |- let (H x, y) = H (H a, b) in c == let (H x, y) = H d in c @ L]]$,
and the left-hand side reduces to $[[c {a / x, b / y}]]$,
while the right-hand side again does not necessarily reduce to something indistinguishable from it.
Even if the context were empty, the simulation property still cannot be proven
without knowing \apriori that a closed proof of equality $[[p]]$ normalizes to $[[refl]]$,
or that a closed term of a pair type normalizes to a pair.

The one exception is the eliminator for the empty type,
whose scrutinee may be well typed at any arbitrary level in \rref{Wt-Absurd}.
Since $[[absurd b]]$ has no reduction rules beyond the congruence \rref{P-Absurd},
ignoring $[[b]]$ in \rref{I-Absurd} will not violate the simulation property.
