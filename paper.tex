\section{Notes}

The level of scrutinees must be bounded by the level of the overall eliminator.
In particular, in \rref{Wt-J}, the level of an equality proof $[[p]]$
must be bounded by that of its elimination $[[J c p]]$,
and in \rref{Wt-Let}, the level $[[l1]]$ of a pair $[[a]]$
must be bounded by that of its elimination $[[let (l0 x, y) = l1 a in b]]$.
The bound is required to prove the simulation property (\cref{lem:simulation}).
Suppose that scrutinees could instead be typed at any level.
First, the indistinguishability rules need to be updated
so that scrutinees are indistinguishable if their level is higher
than that of the overall eliminator, just like in \rref{I-App}.
The modified premises are highlighted in \hlcolor{} below.
%
\begin{mathpar}
  \setlength{\fboxsep}{1.5pt}
  \infer[I-J-bad]{
    \mathcolorbox{\hlcolor}{[[E |- p0 == p1 @ l l0]]} \\
    [[E |- c0 == c1 @ l]]
  }{
    [[E |- J c0 l0 p0 == J c1 l0 p1 @ l]]
  }
  \and
  \infer[I-Let-bad]{
    \mathcolorbox{\hlcolor}{[[E |- a0 == a1 @ l l1]]} \\
    [[(# E ++ x : l0 #) ++ y : l1 |- b0 == b1 @ l]]
  }{
    [[E |- let (l0 x, y) = l1 a0 in b0 == let (l0 x, y) = l1 a1 in b1 @ l]]
  }
\end{mathpar}

As concrete examples, by these alternate rules,
supposing $[[J c p]]$ were well typed in context $[[G]]$
at level $[[L]]$ with $[[G |- p : H a ~ L a]]$,
we can conclude that $[[| G | |- J c refl == J c p @ L]]$.
The problem is that the left-hand side reduces to $[[c]]$,
while the right-hand side does not necessarily reduce to something indistinguishable from $[[c]]$,
thereby violating the simulation property.
The same issue arises with pairs, where we may be able to conclude that
$[[| G | |- let (H x, y) = H (H a, b) in c == let (H x, y) = H d in c @ L]]$,
and the left-hand side reduces to $[[c {a / x, b / y}]]$,
while the right-hand side again does not necessarily reduce to something indistinguishable from it.
Even if the context were empty, the simulation property still cannot be proven
without knowing \apriori that a closed proof of equality $[[p]]$ normalizes to $[[refl]]$,
or that a closed term of a pair type normalizes to a pair.

The one exception is the eliminator for the empty type,
whose scrutinee may be well typed at any arbitrary level in \rref{Wt-Absurd}.
Since $[[absurd b]]$ has no reduction rules beyond the congruence \rref{P-Absurd},
ignoring $[[b]]$ in \rref{I-Absurd} will not violate the simulation property.

For instance, a single substitution $[[{ a / x }]]$ would be semantically well formed
if there exist $i$, $S$ such that
$\interp{[[|D|]]}{[[i]]}{[[A {a / x}]]}{S}$
and $\member{a}{[[l]]}{S}$.}

\begin{definition}[Substitution maps]
  A substitution map $\rho$ is a mapping from variables to terms.
  We use $\dom(\rho)$ to denote the variables in the domain of $\rho$,
  $\rho(x)$ the term to which $\rho$ maps $[[x]]$,
  and $\subst{[[a]]}{\rho}$ to denote the simultaneous substitution
  of the variables $x \in \dom(\rho)$ in $[[a]]$ by $\rho(x)$.
\end{definition}

\begin{definition}[Semantic substitution]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{$\rho$\_ok}}
  A substitution map is semantically well formed
  with respect to a context $[[G]]$ under a context $[[D]]$,
  written $\sws{[[D]]}{\rho}{[[G]]}$,
  if for every $[[x : l A in G]]$,
  there exist $i$, $S$ such that
  $\interp{[[|D|]]}{[[i]]}{\subst{[[A]]}{\rho}}{S}$
  and $\member{\rho(x)}{[[l]]}{S}$.
  Intuitively, $\rho$ replaces the variables listed by $[[G]]$,
  and the terms $\rho$ replaces them by have free variables in $[[D]]$.
\end{definition}

\begin{definition}[Semantic well-typedness]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{SemWt}}
  A term $[[a]]$ is semantically well typed with type $[[A]]$
  at level $[[l]]$ under context $[[G]]$,
  written $\swt{[[G]]}{[[a]]}{[[l]]}{[[A]]}$,
  if for every $\rho$, $[[D]]$,
  given $\sws{[[D]]}{\rho}{[[G]]}$,
  there exist $i$, $S$ such that
  $\interp{[[|D|]]}{[[i]]}{\subst{[[A]]}{\rho}}{S}$
  and $\member{\subst{[[a]]}{\rho}}{[[l]]}{S}$.
\end{definition}

\begin{definition}[Semantic well-formedness]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{SemWff}}
  A context $[[G]]$ is semantically well formed,
  written $\swf{[[G]]}$,
  if for every $[[x : l A in G]]$,
  there exist $j$, $[[l0]]$ such that
  $\swt{[[G]]}{[[A]]}{[[l0]]}{[[Univ j]]}$.
\end{definition}

\begin{lemma}[Semantic substitution, nil]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{$\rho$\_ok\_nil}}
  $\sws{[[nil]]}{\rho}{[[nil]]}$ holds for any $\rho$.
\end{lemma}

\begin{lemma}[Semantic substitution, cons]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{$\rho$\_ok\_cons}}
  Suppose $\sws{[[D]]}{\rho}{[[G]]}$.
  If $\interp{[[|D|]]}{[[i]]}{\subst{[[A]]}{\rho}}{S}$
  and $\member{[[a]]}{[[l]]}{S}$,
  then $\sws{[[D]]}{\substcons{\rho}{[[x]]}{[[a]]}}{[[G ++ x : l A]]}$.
\end{lemma}

\begin{lemma}[Semantic well-formedness, nil]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{SemWff\_nil}}
  $\swf{[[nil]]}$ holds.
\end{lemma}

\begin{lemma}[Semantic well-formedness, cons]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{SemWff\_cons}}
  Suppose $\swf{[[G]]}$.
  If $\swt{[[G]]}{[[A]]}{[[l]]}{[[Univ i]]}$,
  then $\swf{[[G ++ x : l A]]}$.
\end{lemma}

\begin{lemma}[Semantic type inversion]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{SemWt\_Univ}}
  The following are equivalent:
  \begin{itemize}
    \item $\swt{[[G]]}{[[A]]}{[[l]]}{[[Univ i]]}$;
    \item For every $\rho$, $[[D]]$ such that $\sws{[[D]]}{\rho}{[[G]]}$,
      \begin{itemize}
        \item $[[| D | |- A { rho } : l]]$ holds, and
        \item there exists $S$ such that $\interp{[[|D|]]}{[[i]]}{[[A { rho }]]}{S}$.
      \end{itemize}
  \end{itemize}
\end{lemma}

\begin{theorem}[Fundamental theorem]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{soundness}}
  If $[[G |- a : l A]]$ then $\swt{[[G]]}{[[a]]}{[[l]]}{[[A]]}$;
  if $[[|- G]]$ then $\swf{[[G]]}$.
\end{theorem}

\begin{corollary}[Logical consistency]
  \stack{.~}\footnote{\suppl{theories}{soundness.v}{consistency}}
  There are no $[[a]]$, $[[l]]$ such that $[[nil |- a : l Void]]$.
\end{corollary}
